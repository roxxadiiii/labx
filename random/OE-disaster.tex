	\section{Cause --- Glacier Collapse}
	%% ============================================================
	
	\begin{frame}{The Triggering Event}
		\framesubtitle{Rock-Ice Avalanche, 26 August 2026, 08:37 NPT}
		\begin{columns}[T]
			\column{0.55\textwidth}
			\begin{block}{Chain of Events}
				\begin{enumerate}
					\item \textbf{Glacier detaches} from Lirung massif at
					$\sim$5{,}200\,m asl --- 1--1.3\,km wide section
					\item \textbf{Free-falls} $\approx$1{,}200\,m to valley floor
					\item Ice melts from impact $\rightarrow$
					\textbf{massive meltwater surge}
					\item Meltwater + debris $\rightarrow$
					\textbf{long-runout debris flow}
					\item Flow travels $\sim$100\,km downstream in hours
					\item \textbf{Two barrier lakes} form post-collapse
				\end{enumerate}
			\end{block}
			\vspace{4pt}
			{\color{rpMuted}\footnotesize
				USGS: seismic signal $M_s$\,5.2 generated \emph{by} the collapse,
				not preceding it. GFZ recorded $M_w$\,5.7.}
			
			\column{0.42\textwidth}
			%% -------------------------------------------------------
			%% FIGURE 4: glacier_collapse_satellite_map.png
			%% -------------------------------------------------------

					Satellite-derived map: collapse site (yellow reticle,
					$\sim$5{,}200\,m), debris flow path (yellow arrow),
					and inundation zone along Trishuli River}
			\end{center}
		\end{columns}
	\end{frame}

	\section{Early Warning Systems}
	%% ============================================================
	
	\begin{frame}{Warning System --- What Went Wrong}
		\begin{columns}[T]
			\column{0.55\textwidth}
			\footnotesize % Added footnotesize to make text smaller
			\setlength{\itemsep}{2pt} % Tighter spacing between bullets
			\textbf{Pre-existing System}
			\begin{itemize}
				\item China--Nepal joint flash flood early-warning system
				operational for years
				\item \textit{SCMP}: consistently prevented casualties
				in previous events
			\end{itemize}
			
			\vspace{4pt}
			\textbf{Why It Failed in 2026}
			\begin{itemize}
				\item Monitors configured for \textbf{monsoon / perennial
					floods}, not glacier collapses
				\item Upstream stations \textbf{swept away before alerts}
				could be sent
				\item Region was a monitoring \textbf{``blind spot''}
				(Stimson Center)
				\item China's Tibet restrictions
				\textbf{hampered cross-border data sharing}
			\end{itemize}
			
			\column{0.42\textwidth}
			%% -------------------------------------------------------
			%% FIGURE 7: early_warning_monitoring_station.jpg
			%% -------------------------------------------------------
			\begin{center}
				\centering
				\dummyfig{5.3}{3.8}{early\_warning\_monitoring\_station.jpg}
				\vspace{2pt}\par{\color{rpMuted}\tiny \textbf{[early\_warning\_monitoring\_station.jpg]}\\
					River-level monitoring station on the Trishuli
					basin --- stations similar to this were destroyed
					before downstream alerts could be relayed}
			\end{center}
		\end{columns}
	\end{frame}

	\section{Lessons \& Way Forward}
	%% ============================================================
	
	\begin{frame}{Key Lessons}
		\begin{columns}[T]
			\column{0.48\textwidth}
			\begin{block}{Warning System Gaps}
				\begin{itemize}
					\item Systems built for \textbf{monsoon floods} are blind
					to \textbf{glacier collapse} signatures
					\item Upstream sensors must be hardened against
					the events they monitor
					\item Political barriers to cross-border data sharing
					\textbf{directly cost lives}
				\end{itemize}
			\end{block}
			
			\column{0.48\textwidth}
			\begin{block}{Climate Resilience}
				\begin{itemize}
					\item Permafrost degradation is
					\textbf{accelerating} across the Himalayas
					\item Glacier collapses are rare but
					\textbf{catastrophic}; low-probability events
					must be planned for
					\item Satellite glacier-velocity monitoring
					\textbf{can provide weeks of lead time}
				\end{itemize}
			\end{block}
		\end{columns}
		
		\vspace{8pt}
		\begin{exampleblock}{Way Forward}
			\centering
			Multi-hazard EWS \textbullet\
			Cross-border real-time monitoring \textbullet\
			Glacier-velocity satellite surveillance \textbullet\
			Climate-adaptive infrastructure \textbullet\
			Regional disaster financing
		\end{exampleblock}
	\end{frame}
	
	\begin{frame}{Recommendations}
		\begin{enumerate}
			\item \textbf{Upgrade Early Warning Systems} --- integrate
			glacier-specific sensors (seismic, GPS, SAR) alongside
			conventional water-level gauges
			\item \textbf{Strengthen Cross-Border Cooperation} --- China--Nepal
			real-time data-sharing protocol with defined SLAs
			\item \textbf{Satellite-Based Glacier Monitoring} --- near-daily
			optical/SAR imagery; velocity anomaly auto-alerts
			\item \textbf{Community-Level Preparedness} --- vertical evacuation
			routes, community wardens, multilingual SMS systems
			\item \textbf{Climate Finance} --- Nepal and frontline nations must
			receive dedicated adaptation funds at COP
			\item \textbf{Rethink Infrastructure Siting} --- hydropower projects
			and border crossings must account for glacier-collapse
			run-out zones
		\end{enumerate}
	\end{frame}
